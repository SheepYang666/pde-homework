\noindent
已知 $\dfrac{\mathrm{d}u}{\mathrm{d}t}=\sin u$，$u(0)=0.1$，$0\le t\le 10$。
求 $t=10$ 时的数值解，\linebreak
并比较 $N=100,200,400,800$ 时数值解与精确解的误差 $E=\text{精确解}-\text{数值解}$。

\begin{enumerate}
  \item 求$\frac{\mathrm{d} u}{\mathrm{d} t} = \sin {u}$解析解
    \[
      \int \frac{\mathrm{d}u}{\sin u} = \int \mathrm{d}t
      \Longrightarrow
      \ln\left|\tan\frac{u}{2}\right| = t + C
    \]

  \item 带入$u(0)=0.1$
    \[
      C = \ln{(\tan{0.05})}
      \Longrightarrow
      u(t)=2\arctan\left(e^{t}\tan\frac{u_{0}}{2}\right)=2\arctan\left(e^{t}\tan 0.05\right)
    \]
    \[
      u(10) = 2\arctan\left(e^{10}\tan 0.05\right) = 3.139778170480394
    \]
  
  \item 采用二阶改进欧拉法逐步计算

    每步先用欧拉公式\textbf{预估}
    \[
      \tilde{u}_{n+1} = u_n + h\,f(t_n,u_n),
    \]
    再用梯形公式\textbf{校正}：
    \[
      u_{n+1} = u_n + \frac{h}{2}\Bigl[f(t_n,u_n) + f\bigl(t_{n+1},\,\tilde{u}_{n+1}\bigr)\Bigr].
    \]
    对于 $f(t,u)=\sin u$，$u(0)=0.1$，取步长 $h=10/N$，从 $t=0$ 逐步计算到 $t=10$。核心迭代代码如下：

    \begin{lstlisting}
double u = 0.1;                     // 初值
const double h = 10.0 / N;          // 步长
for (int n = 0; n < N; ++n) {
  const double t = n * h;
  const double up = u + h * sin(u);              // 预估：欧拉步
  u = u + 0.5 * h * (sin(u) + sin(up));          // 校正：梯形公式
}
    \end{lstlisting}

    取 $N=100,\,200,\,400,\,800$，$t=10$ 处的数值解与误差 $E=\text{精确解}-\text{数值解}$ 列于表~\ref{tab:euler}。

    \begin{table}[htbp]
      \centering
      \caption{$t=10$ 处改进欧拉法的数值解与误差}
      \label{tab:euler}
      \begin{tabular}{ccccc}
        \toprule
        $N$ & 步长 $h$ & 数值解 $u_N(10)$ & 精确解 $u(10)$ & 误差 $E$ \\
        \midrule
        $100$ & $0.1$    & $3.1397497508$ & $3.1397781705$ & $2.8420\times 10^{-5}$ \\
        $200$ & $0.05$   & $3.1397712333$ & $3.1397781705$ & $6.9372\times 10^{-6}$ \\
        $400$ & $0.025$  & $3.1397764537$ & $3.1397781705$ & $1.7168\times 10^{-6}$ \\
        $800$ & $0.0125$ & $3.1397777433$ & $3.1397781705$ & $4.2723\times 10^{-7}$ \\
        \bottomrule
      \end{tabular}
    \end{table}

    由上述表可知 $N$ 每加倍一次，误差约缩小为原来的 $1/4$（相邻误差比依次为 $4.097$、$4.041$、$4.019$），与二阶方法的理论精度 $O(h^2)$ 相符合。
\end{enumerate}
